#### 1. **Quality-First Approach**
```python
# Don't treat all reviews equally
def select_best_reviews(reviews, limit=5):
    """Select highest quality reviews for processing."""
    
    # Priority 1: Featured reviews (curated by AllRecipes)
    featured = [r for r in reviews if r.is_featured and r.has_modification]
    
    # Priority 2: High-rated reviews (4-5 stars)
    high_rated = [r for r in reviews if r.rating >= 4 and r.has_modification]
    
    # Priority 3: Sort by rating
    candidates = featured or high_rated or reviews
    sorted_reviews = sorted(candidates, key=lambda r: r.rating or 0, reverse=True)
    
    return sorted_reviews[:limit]
```

#### 2. **Complete Extraction (Not Partial)**
```python
# Extract ALL modifications from a review, not just one
def extract_all_modifications(review_text, recipe):
    """Extract all discrete modifications from review."""
    
    prompt = f"""
    Review: "{review_text}"
    
    Extract ALL discrete modifications as a JSON array.
    Each separate change should be a separate object.
    
    Example:
    Review: "I used less sugar, added vanilla, and chilled the dough"
    Output: [
        {{"modification_type": "quantity_adjustment", ...}},
        {{"modification_type": "addition", ...}},
        {{"modification_type": "technique_change", ...}}
    ]
    """
    
    response = llm.generate(prompt)
    modifications = [ModificationObject(**m) for m in json.loads(response)]
    
    return modifications  # Returns LIST, not single object
```

#### 3. **Validation & Error Detection**
```python
def validate_modification(modification, recipe):
    """Validate that modification can actually be applied."""
    
    warnings = []
    
    for edit in modification.edits:
        # Check if target exists
        target_content = recipe.ingredients if edit.target == "ingredients" else recipe.instructions
        match, score = fuzzy_match(edit.find, target_content)
        
        if not match:
            warnings.append(f"Cannot find '{edit.find}' in {edit.target}")
        elif score < 0.8:
            warnings.append(f"Low confidence match for '{edit.find}' (score: {score:.2f})")
        
        # Check for required fields
        if edit.operation == "replace" and not edit.replace:
            warnings.append(f"Replace operation missing replacement text")
    
    return len(warnings) == 0, warnings
```

#### 4. **Robust Application (No Silent Failures)**
```python
def apply_edit_robust(edit, recipe_content):
    """Apply edit with explicit success tracking."""
    
    match, index, score = find_best_match(edit.find, recipe_content)
    
    if not match:
        logger.warning(f"Could not find '{edit.find}' (best score: {score:.2f})")
        return recipe_content, None  # Explicit failure
    
    logger.info(f"Matched '{edit.find}' to '{match}' (score: {score:.2f})")
    
    # Use the MATCHED string, not the target
    if edit.operation == "replace":
        # Option 1: Replace within the matched line
        new_line = match.replace(edit.find, edit.replace)
        recipe_content[index] = new_line
        
        # Option 2: Replace the entire matched line (more robust)
        # recipe_content[index] = edit.replace
        
        change = ChangeRecord(
            type="ingredient",
            from_text=match,
            to_text=new_line,
            operation="replace",
            confidence=score  # ← Track confidence!
        )
        
        return recipe_content, change
    
    # ... handle other operations
```

#### 5. **Comprehensive Evaluation**
```python
class EvaluationFramework:
    """Measure pipeline performance."""
    
    def __init__(self):
        self.test_cases = self.load_gold_dataset()
    
    def load_gold_dataset(self):
        """Load hand-labeled ground truth."""
        return [
            {
                "review": "I used half cup of sugar and 1.5 cups brown sugar",
                "expected_modifications": [
                    {
                        "type": "quantity_adjustment",
                        "edits": [
                            {"find": "1 cup white sugar", "replace": "0.5 cup white sugar"},
                            {"find": "1 cup brown sugar", "replace": "1.5 cups brown sugar"}
                        ]
                    }
                ]
            },
            # ... 19+ more test cases
        ]
    
    def evaluate(self, pipeline):
        """Run evaluation."""
        results = []
        
        for test_case in self.test_cases:
            predicted = pipeline.extract(test_case["review"])
            expected = test_case["expected_modifications"]
            
            precision = self.calculate_precision(predicted, expected)
            recall = self.calculate_recall(predicted, expected)
            f1 = 2 * (precision * recall) / (precision + recall) if (precision + recall) > 0 else 0
            
            results.append({
                "test_case": test_case,
                "precision": precision,
                "recall": recall,
                "f1": f1
            })
        
        return {
            "avg_precision": np.mean([r["precision"] for r in results]),
            "avg_recall": np.mean([r["recall"] for r in results]),
            "avg_f1": np.mean([r["f1"] for r in results]),
            "details": results
        }
    
    def calculate_precision(self, predicted, expected):
        """How many predicted edits are correct?"""
        correct = sum(1 for p in predicted if p in expected)
        return correct / len(predicted) if predicted else 0
    
    def calculate_recall(self, predicted, expected):
        """How many expected edits were found?"""
        correct = sum(1 for e in expected if e in predicted)
        return correct / len(expected) if expected else 0
```
